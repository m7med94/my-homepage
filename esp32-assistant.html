from fastapi import FastAPI, Header, HTTPException, status
from pydantic import BaseModel, Field
from typing import Optional
import time

app = FastAPI(title="ESP32 Voice Assistant Telemetry Service")

# 1. Pydantic Payload Validation Schema
class DeviceDataPayload(BaseModel):
    device_id: str = Field(..., example="mo-project-c3")
    category: str = Field(..., example="user_request")
    data: str = Field(..., example="Turn on air conditioning")
    timestamp: Optional[int] = Field(default_factory=lambda: int(time.time()))

# 2. Ingestion Endpoint
@app.post("/api/v1/device/data", status_code=status.HTTP_201_CREATED)
async def receive_device_data(
    payload: DeviceDataPayload,
    authorization: Optional[str] = Header(None)
):
    # Optional Security Check
    if authorization and not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Invalid token format")
    
    # Process incoming telemetry / voice event
    print(f"[ESP32 INGEST] Device: {payload.device_id} | Category: {payload.category} | Msg: {payload.data}")
    
    return {
        "status": "success",
        "received_at": int(time.time()),
        "device_id": payload.device_id,
        "processed_category": payload.category
    }
import express, { Request, Response } from 'express';

const app = express();
app.use(express.json());

interface DevicePayload {
  device_id: string;
  category: string;
  data: string;
  timestamp?: number;
}

app.post('/api/v1/device/data', (req: Request<{}, {}, DevicePayload>, res: Response) => {
  const { device_id, category, data, timestamp } = req.body;
  const authHeader = req.headers.authorization;

  if (!device_id || !category || data === undefined) {
    return res.status(400).json({ error: 'Missing required payload fields' });
  }

  console.log(`[ESP32] Node ${device_id} [${category}]: ${data}`);

  return res.status(201).json({
    status: 'success',
    device_id,
    received_timestamp: timestamp || Math.floor(Date.now() / 1000)
  });
});

app.listen(8000, () => console.log('Server listening on port 8000'));
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* serverUrl = "http://YOUR_SERVER_IP/api/v1/device/data";
const char* deviceApiKey = "secret-device-token-123";

void sendTelemetryPayload(String category, String dataMsg) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Authorization", String("Bearer ") + deviceApiKey);

    StaticJsonDocument<256> doc;
    doc["device_id"] = "mo-project-c3";
    doc["category"] = category;
    doc["data"] = dataMsg;
    doc["timestamp"] = (unsigned long)time(NULL);

    String jsonString;
    serializeJson(doc, jsonString);

    int httpCode = http.POST(jsonString);
    Serial.printf("Payload sent, response code: %d\n", httpCode);
    http.end();
  }
}
