2. Endpoint Specification

REST Contract
POST /api/v1/device/data Ingest device telemetry
GET /api/v1/device/data?device_id=...&category=...&limit=50 Query historical records
GET /health Service health check

HTTP Request Headers

Header Type / Format Requirement
Content-Type application/json Required
Authorization Bearer <OPTIONAL_DEVICE_API_KEY> Optional (Device Authentication)

Incoming Request Payload (JSON Schema)

{
  "device_id": "string",       // e.g. "mo-project-c3" or MAC address (Required)
  "category": "string",        // e.g. "general", "alert", "user_request", "telemetry" (Required)
  "data": "string",            // The message, sensor reading, or JSON payload from assistant (Required)
  "timestamp": 1723896000      // Optional Unix timestamp (seconds or ms)
}

3. Required Server Features & Architecture

🛡️ Input Validation

Strict payload schema checking (Pydantic/Zod). Gracefully handles large text strings, UTF-8 voice transcripts, or escaped JSON within the data field.

💾 Database Persistence

Persistent storage (SQLite/PostgreSQL) with fields: id (UUID/PK), device_id (Indexed), category (Indexed), payload_data, client_ip, and created_at.

📊 Logging & Monitoring

Structured console and file logging for every request with latency benchmarks, device ID, and health check route GET /health.

🌐 CORS & 0.0.0.0 Binding

Bound to 0.0.0.0 (not just 127.0.0.1) so ESP32 devices on your local Wi-Fi router can reach the server immediately without connection refusals.

# Pydantic Ingestion Schema
class DevicePayload(BaseModel):
    device_id: str = Field(..., description="Device ID or MAC", example="mo-project-c3")
    category: str = Field(..., description="Payload category", example="user_request")
    data: str = Field(..., description="Message or JSON string", example="Turn on air conditioning")
    timestamp: Optional[int] = Field(None, description="Optional timestamp")

# Health Check Route
@app.get("/health")
def health_check():
    return {"status": "healthy", "timestamp": datetime.now(timezone.utc).isoformat()}

# POST Telemetry Ingestion Route
@app.post("/api/v1/device/data", status_code=status.HTTP_201_CREATED)
async def ingest_device_data(
    payload: DevicePayload,
    request: Request,
    authorization: Optional[str] = Header(None)
):
    start_time = time.time()
    client_ip = request.client.host if request.client else "unknown"
    entry_id = str(uuid.uuid4())

    # Persist to SQLite
    with sqlite3.connect("esp32_telemetry.db") as conn:
        conn.execute(
            "INSERT INTO telemetry_logs (id, device_id, category, payload_data, client_ip) VALUES (?, ?, ?, ?, ?)",
            (entry_id, payload.device_id, payload.category, payload.data, client_ip)
        )

    duration_ms = round((time.time() - start_time) * 1000, 2)
    print(f"[{datetime.now().strftime('%H:%M:%S')}] [201] Device:{payload.device_id} Cat:{payload.category} IP:{client_ip} ({duration_ms}ms)")

    return {
        "status": "success",
        "message": "Data received and processed",
        "received_at": datetime.now(timezone.utc).isoformat()
    }

# GET Query Records Route
@app.get("/api/v1/device/data")
def query_device_data(
    device_id: Optional[str] = Query(None),
    category: Optional[str] = Query(None),
    limit: int = Query(50, le=200)
):
    query = "SELECT id, device_id, category, payload_data, client_ip, created_at FROM telemetry_logs WHERE 1=1"
    params = []
    if device_id:
        query += " AND device_id = ?"
        params.append(device_id)
    if category:
        query += " AND category = ?"
        params.append(category)
    query += " ORDER BY created_at DESC LIMIT ?"
    params.append(limit)

    with sqlite3.connect("esp32_telemetry.db") as conn:
        conn.row_factory = sqlite3.Row
        rows = conn.execute(query, params).fetchall()

    return {"count": len(rows), "results": [dict(r) for r in rows]}

// server.js — Express.js + SQLite Implementation
const express = require('express');
const cors = require('cors');
const Database = require('better-sqlite3');
const { v4: uuidv4 } = require('uuid');

const app = express();
const db = new Database('esp32_telemetry.db');

app.use(cors());
app.use(express.json({ limit: '2mb' }));

// Init Table
db.exec(`
  CREATE TABLE IF NOT EXISTS telemetry_logs (
    id TEXT PRIMARY KEY,
    device_id TEXT NOT NULL,
    category TEXT NOT NULL,
    payload_data TEXT NOT NULL,
    client_ip TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP
  );
  CREATE INDEX IF NOT EXISTS idx_device ON telemetry_logs (device_id);
`);

// Health Check
app.get('/health', (req, res) => {
  res.json({ status: 'healthy', timestamp: new Date().toISOString() });
});

// POST Ingestion
app.post('/api/v1/device/data', (req, res) => {
  const { device_id, category, data } = req.body;

  if (!device_id || !category || data === undefined) {
    return res.status(422).json({
      status: 'error',
      message: "Missing required field: 'data', 'device_id' or 'category'"
    });
  }

  const id = uuidv4();
  const clientIp = req.ip || req.connection.remoteAddress;

  const stmt = db.prepare(
    'INSERT INTO telemetry_logs (id, device_id, category, payload_data, client_ip) VALUES (?, ?, ?, ?, ?)'
  );
  stmt.run(id, device_id, category, typeof data === 'string' ? data : JSON.stringify(data), clientIp);

  return res.status(201).json({
    status: 'success',
    message: 'Data received and processed',
    received_at: new Date().toISOString()
  });
});

// Bind 0.0.0.0 for LAN access
app.listen(8000, '0.0.0.0', () => {
  console.log('ESP32 Telemetry Gateway listening on http://0.0.0.0:8000');
});

#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>

const char* serverUrl = "http://192.168.1.100:8000/api/v1/device/data";
const char* deviceApiKey = "secret-device-token-123";

void sendVoiceAssistantEvent(String category, String transcriptOrData) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(serverUrl);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Authorization", String("Bearer ") + deviceApiKey);

    StaticJsonDocument<512> doc;
    doc["device_id"] = "mo-project-c3";
    doc["category"] = category;
    doc["data"] = transcriptOrData;
    doc["timestamp"] = (unsigned long)time(NULL);

    String jsonString;
    serializeJson(doc, jsonString);

    int httpResponseCode = http.POST(jsonString);
    String response = http.getString();
    Serial.printf("[HTTP %d] Server Response: %s\n", httpResponseCode, response.c_str());
    http.end();
  }
}
